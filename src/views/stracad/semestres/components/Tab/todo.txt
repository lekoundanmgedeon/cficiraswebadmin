<script setup>
import { ref, onMounted, watch } from 'vue';
import axios from 'axios'; // Ou votre client API personnalisé (ex: import api from '@/services/api')

// État de sélection de l'année
const selectedPeriod = ref('2025-2026');
const loading = ref(false);

// Données réactives mappées sur le template
const kpis = ref({
  volume_horaire_global: 0,
  taux_assiduite_global: 92.4,
  corps_enseignant_total: 0,
  total_ues: 0
});

const semesterStats = ref([]);

const typology = ref({
  fondamental: { heures: 0, pourcentage: '0' },
  optionnel: { heures: 0, pourcentage: '0' },
  transversal: { heures: 0, pourcentage: '0' }
});

// Fonction de chargement des données depuis l'API unifiée
const loadDashboardAnalytics = async () => {
  loading.value = true;
  try {
    // Remplacer par l'URL exacte de votre API ou votre instance Axios
    const res = await axios.get(`/api/academique/semestres/analytics/dashboard?period=${selectedPeriod.value}`);
    
    if (res.data && res.data.success) {
      const data = res.data.data;
      
      // 1. Liaison des cartes KPI
      kpis.value = data.kpis;
      
      // 2. Liaison de la matrice (tableau principal)
      semesterStats.value = data.matrix.map((row, idx) => ({
        id: row.semestre_id || idx,
        code: row.semestre_code,
        niveau: row.niveau,
        filiere: row.filiere,
        nbUes: row.nb_ues,
        totalHeures: row.total_heures,
        moyenneGenerale: row.moyenne_generale,
        statutMaquette: row.statut_maquette
      }));

      // 3. Liaison de la typologie des cours (Barres de progression latérales)
      // Réinitialisation par défaut
      typology.value = {
        fondamental: { heures: 0, pourcentage: '0' },
        optionnel: { heures: 0, pourcentage: '0' },
        transversal: { heures: 0, pourcentage: '0' }
      };

      data.typology.forEach(item => {
        const type = item.type_module?.toLowerCase();
        if (type.includes('fond') || type.includes('uef')) {
          typology.value.fondamental = { heures: item.volume_heures, pourcentage: item.pourcentage };
        } else if (type.includes('opt') || type.includes('spec')) {
          typology.value.optionnel = { heures: item.volume_heures, pourcentage: item.pourcentage };
        } else {
          typology.value.transversal = { heures: item.volume_heures, pourcentage: item.pourcentage };
        }
      });
    }
  } catch (error) {
    console.error("Erreur lors de la récupération des analytics :", error);
  } finally {
    loading.value = false;
  }
};

// Recharger les données dès que l'utilisateur change d'année académique dans le select
watch(selectedPeriod, () => {
  loadDashboardAnalytics();
});

// Chargement initial au montage du composant
onMounted(() => {
  loadDashboardAnalytics();
});
</script>
